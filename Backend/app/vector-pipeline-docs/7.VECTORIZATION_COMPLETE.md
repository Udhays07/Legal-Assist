## 🔄 How It Works

### Automatic Workflow

```
┌─────────────────────────────────────────────────────────────┐
│                    DOCUMENT LIFECYCLE                        │
└─────────────────────────────────────────────────────────────┘

CREATE DOCUMENT (POST /documents)
    │
    ├─► Save document to database
    │
    ├─► Generate 768-dim embedding vector
    │
    └─► Store in document_embeddings table
         ✓ Ready for semantic search!

UPDATE DOCUMENT (PUT /documents/{id})
    │
    ├─► Update document in database
    │
    ├─► Check if content was modified
    │
    └─► If yes: Regenerate embedding
         If no: Skip (optimization)

DELETE DOCUMENT (DELETE /documents/{id})
    │
    ├─► Soft delete document (set deleted_at)
    │
    └─► Delete associated embedding
         ✓ Clean up complete!
```

    COUNT(DISTINCT d.id) as total_documents,
    COUNT(DISTINCT e.document_id) as documents_with_embeddings,
    ROUND(COUNT(DISTINCT e.document_id)::numeric / COUNT(DISTINCT d.id) * 100, 2) as coverage_pct

LEFT JOIN document_embeddings e ON d.id = e.document_id
WHERE d.deleted_at IS NULL;

## 🎨 Next Steps: Semantic Search

Now that embeddings are generated, you can implement semantic search:

### Example: Find Similar Documents

```python
from app.services.embedding_service import generate_embedding
from sqlalchemy import text
from app.core.database import SessionLocal

db = SessionLocal()

# Generate query embedding
query = "contract termination clause"
query_embedding = generate_embedding(query)

# Find top 10 similar documents
results = db.execute(text("""
    SELECT 
        d.id,
        d.title,
        d.content,
        1 - (e.embedding <=> :query_embedding::vector) as similarity_score
    FROM documents d
    JOIN document_embeddings e ON d.id = e.document_id
    WHERE d.deleted_at IS NULL
    ORDER BY e.embedding <=> :query_embedding::vector
    LIMIT 10
"""), {"query_embedding": query_embedding}).fetchall()

for doc in results:
    print(f"{doc.title}: {doc.similarity_score:.2%} match")

db.close()
```

### Create Semantic Search Endpoint

```python
# Add to app/api/document.py

@router.post("/search/semantic")
def semantic_search(
    query: str,
    limit: int = 10,
    db: Session = Depends(get_db)
):
    """Search documents by semantic similarity."""
    query_embedding = generate_embedding(query)
    
    results = db.execute(text("""
        SELECT 
            d.id, d.title, d.content,
            1 - (e.embedding <=> :query) as similarity
        FROM documents d
        JOIN document_embeddings e ON d.id = e.document_id
        WHERE d.deleted_at IS NULL
        ORDER BY e.embedding <=> :query
        LIMIT :limit
    """), {"query": query_embedding, "limit": limit}).fetchall()
    
    return [
        {
            "id": str(r.id),
            "title": r.title,
            "content": r.content[:200] + "...",
            "similarity": round(r.similarity, 4)
        }
        for r in results
    ]
```
